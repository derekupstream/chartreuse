    ```typescript
    export type IssueInput = {
      issueType: string;
      severity: 'error' | 'warning';
      entity: string;       // 'Project' | 'SingleUseLineItem' | 'ReusableLineItem'
    ```

    **The 9 check functions (implement each as a standalone async function):**

    Error checks (severity: 'error'):
    1. `checkMissingUSState()` — `prisma.project.findMany({ where: { USState: null }, select: { id: true } })` → entity: `'Project'`, issueType: `'missing_us_state'`

    Warning checks (severity: 'warning'):
    5. `checkReturnRateOver100()` — `prisma.reusableLineItem.findMany({ where: { annualRepurchasePercentage: { gt: 100 } }, select: { id: true, annualRepurchasePercentage: true } })` → entity: `'ReusableLineItem'`, issueType: `'return_rate_over_100'`, details: `{ annualRepurchasePercentage: row.annualRepurchasePercentage }`
    6. `checkReusableNegativeCaseCost()` — `prisma.reusableLineItem.findMany({ where: { caseCost: { lt: 0 } }, select: { id: true, caseCost: true } })` → entity: `'ReusableLineItem'`, issueType: `'reusable_negative_case_cost'`, details: `{ caseCost: row.caseCost }`
    7. `checkUnrealisticCaseCost()` — `prisma.reusableLineItem.findMany({ where: { caseCost: { gt: 1_000_000_000 } }, select: { id: true, caseCost: true } })` → entity: `'ReusableLineItem'`, issueType: `'unrealistic_case_cost'`, details: `{ caseCost: row.caseCost }`
    8. `checkSingleUseNegativeCaseCost()` — `prisma.singleUseLineItem.findMany({ where: { caseCost: { lt: 0 } }, select: { id: true, caseCost: true } })` → entity: `'SingleUseLineItem'`, issueType: `'single_use_negative_case_cost'`, details: `{ caseCost: row.caseCost }`
    9. `checkNegativeQuantity()` — `prisma.reusableLineItem.findMany({ where: { casesPurchased: { lt: 0 } }, select: { id: true, casesPurchased: true } })` → entity: `'ReusableLineItem'`, issueType: `'negative_quantity'`, details: `{ casesPurchased: row.casesPurchased }`

    Note on issueType naming: Use entity-prefixed strings for the two `negative_case_cost` checks (`reusable_negative_case_cost` and `single_use_negative_case_cost`) to guarantee uniqueness of the `@@unique([issueType, entityId])` key across entity types.

    **CHECKS array and runner:**
    ```typescript
    const CHECKS: CheckFn[] = [
      checkMissingUSState,
      checkMissingSingleUseItems,
      checkMissingReusableItems,
      checkZeroUnitLineItems,
      checkReturnRateOver100,
      checkReusableNegativeCaseCost,
      checkUnrealisticCaseCost,
      checkSingleUseNegativeCaseCost,
      checkNegativeQuantity,
    ];

    export async function runDataHealthScan(): Promise<IssueInput[]> {
      const results = await Promise.all(CHECKS.map(fn => fn()));
      return results.flat();
    }
    ```

    **File 1: pages/api/admin/data-health/scan.ts** (POST — run scan, upsert, return issues)
    ```typescript
    import type { NextApiResponse } from 'next';
    import type { NextApiRequestWithUser } from 'lib/middleware/getUser';
    import { handlerWithUser } from 'lib/middleware/handler';
    import { checkIsUpstream } from 'lib/middleware/requireUpstream';
    import prisma from 'lib/prisma';
    import { runDataHealthScan } from 'lib/admin/dataHealthScan';

    export default handlerWithUser()
      .post(async (req: NextApiRequestWithUser, res: NextApiResponse) => {
        const isUpstream = await checkIsUpstream(req.user.orgId);
        if (!isUpstream) return res.status(403).json({ error: 'Forbidden' });

        const issues = await runDataHealthScan();

        await Promise.all(
          issues.map(issue =>
            prisma.dataHealthIssue.upsert({
              where: { issueType_entityId: { issueType: issue.issueType, entityId: issue.entityId } },
              create: {
                issueType: issue.issueType,
                severity: issue.severity,
                entity: issue.entity,
                entityId: issue.entityId,
                details: issue.details ?? undefined,
                status: 'open'
              },
              update: {
                severity: issue.severity,
                details: issue.details ?? undefined
                // Do NOT include status — preserve acknowledged/resolved state
              }
            })
          )
        );

        const all = await prisma.dataHealthIssue.findMany({
          where: { status: { not: 'resolved' } },
          orderBy: [{ severity: 'asc' }, { createdAt: 'desc' }]
        });
        res.json(all);
      });
    ```

    **File 2: pages/api/admin/data-health/issues.ts** (GET — list non-resolved issues)
    ```typescript
    import type { NextApiResponse } from 'next';
    import type { NextApiRequestWithUser } from 'lib/middleware/getUser';
    import { handlerWithUser } from 'lib/middleware/handler';
    import { checkIsUpstream } from 'lib/middleware/requireUpstream';
    import prisma from 'lib/prisma';

    export default handlerWithUser()
      .get(async (req: NextApiRequestWithUser, res: NextApiResponse) => {
        const isUpstream = await checkIsUpstream(req.user.orgId);
        if (!isUpstream) return res.status(403).json({ error: 'Forbidden' });

        const { status } = req.query as { status?: string };
        const issues = await prisma.dataHealthIssue.findMany({
          where: status ? { status } : { status: { not: 'resolved' } },
          orderBy: [{ severity: 'asc' }, { createdAt: 'desc' }]
        });
        res.json(issues);
      });
    ```

    **File 3: pages/api/admin/data-health/issues/[id].ts** (PATCH — acknowledge or resolve)
    ```typescript
    import type { NextApiResponse } from 'next';
    import type { NextApiRequestWithUser } from 'lib/middleware/getUser';
    import { handlerWithUser } from 'lib/middleware/handler';
    import { checkIsUpstream } from 'lib/middleware/requireUpstream';
    import prisma from 'lib/prisma';

    export default handlerWithUser()
      .patch(async (req: NextApiRequestWithUser, res: NextApiResponse) => {
        const isUpstream = await checkIsUpstream(req.user.orgId);
        if (!isUpstream) return res.status(403).json({ error: 'Forbidden' });

        const { id } = req.query as { id: string };
        const { status, note } = req.body as { status: string; note?: string };

        if (!status || !['acknowledged', 'resolved'].includes(status))
          return res.status(400).json({ error: 'status must be acknowledged or resolved' });

        const updated = await prisma.dataHealthIssue.update({
          where: { id },
          data: {
            status,
            note: note || null,
            ...(status === 'acknowledged' && {
              acknowledgedAt: new Date(),
              acknowledgedByUserId: req.user.id
            })
          }
        });
        res.json(updated);
      });
    ```
